# --- Core Configuration ---
PROJECT_DIR_NAME="kaleidoscope_project_organized"

FORCE_MODE=false

# --- Helper Functions ---
log() {
    local level="$1"
    local message="$2"
    local timestamp=$(date "+%Y-%m-%d %H:%M:%S")
    local formatted_message="[$timestamp] [$level] $message"
    
    case "$level" in
        "INFO")  echo -e "\033[0;32m$formatted_message\033[0m" ;;
        "WARN")  echo -e "\033[0;33m$formatted_message\033[0m" >&2 ;;
        "ERROR") echo -e "\033[0;31m$formatted_message\033[0m" >&2 ;;
        "DEBUG") 
            if [[ "$VERBOSE" == "true" ]]; then
                echo -e "\033[0;36m$formatted_message\033[0m"
            fi
            ;;
        *) echo "$formatted_message" ;;
    esac
    
    echo "$formatted_message" >> "$LOG_FILE"
}

log_info() { log "INFO" "$1"; }
log_warning() { log "WARN" "$1"; }
log_error() { log "ERROR" "$1"; }
log_debug() { log "DEBUG" "$1"; }

error_exit() {
    log_error "$1"
    log_error "Setup failed. Review '$LOG_FILE' for detailed information."
    
    case "$2" in
        "python_version")
            log_info "Suggestion: Install Python $MIN_PYTHON_VERSION+ using your system package manager"
            log_info "  Ubuntu/Debian: sudo apt install python3.10"
            log_info "  Fedora/RHEL: sudo dnf install python3.10"
            log_info "  macOS: brew install python@3.10"
            ;;
        "pip")
            log_info "Suggestion: Install pip for Python"
            log_info "  Ubuntu/Debian: sudo apt install python3-pip"
            log_info "  Fedora/RHEL: sudo dnf install python3-pip"
            log_info "  macOS: python3 -m ensurepip --upgrade"
            ;;
        "venv")
            log_info "Suggestion: Install venv module"
            log_info "  Ubuntu/Debian: sudo apt install python3-venv"
            log_info "  Fedora/RHEL: sudo dnf install python3-venv"
            ;;
        "dependencies")
            log_info "Suggestion: Check '$REQUIREMENTS_FILE' for errors"
            log_info "  Try: pip install --upgrade pip setuptools wheel"
            ;;
        "port_conflict")
            log_info "Suggestion: Manually configure ports in '$CONFIG_FILE_NAME'"
            ;;
        *)
            log_info "Try running with --force flag for aggressive setup"
            ;;
    esac
    
    exit 1
}

command_exists() {
    command -v "$1" &> /dev/null
}

auto_detect_tools() {
    log_info "Auto-detecting installed tools..."
    
    if [ -z "$GHIDRA_PATH" ]; then
        for path in \
            "/opt/ghidra" \
            "/usr/local/ghidra" \
            "$HOME/ghidra" \
            "$HOME/tools/ghidra" \
            "$(pwd)/ghidra"; do
            if [ -d "$path" ]; then
                server_script=$(find "$path" -name "ghidraSvr" -type f -executable 2>/dev/null | head -n 1)
                if [ -n "$server_script" ]; then
                    GHIDRA_PATH="$server_script"
                    log_info "Found Ghidra server at: $GHIDRA_PATH"
                    break
                fi
            fi
        done
    fi
    
    LLAMA_SERVER_FOUND_PATH=""
    if command_exists "$LLAMA_SERVER_CMD"; then
        LLAMA_SERVER_FOUND_PATH=$(command -v "$LLAMA_SERVER_CMD")
    elif [ -x "$LOCAL_BIN_DIR/$LLAMA_SERVER_CMD" ]; then
        LLAMA_SERVER_FOUND_PATH="$LOCAL_BIN_DIR/$LLAMA_SERVER_CMD"
    elif [ -x "$LLAMA_CPP_BUILD_DIR/$LLAMA_SERVER_CMD" ]; then
        LLAMA_SERVER_FOUND_PATH="$LLAMA_CPP_BUILD_DIR/$LLAMA_SERVER_CMD"
    fi
    if [ -n "$LLAMA_SERVER_FOUND_PATH" ]; then
        LLAMA_SERVER_CMD="$LLAMA_SERVER_FOUND_PATH"
        log_info "Found Llama.cpp server at: $LLAMA_SERVER_CMD"
    fi
    
    if command_exists nvidia-smi && nvidia-smi -L 2>/dev/null | grep -q "GPU"; then
        GPU_SUPPORT=true
        log_info "NVIDIA GPU detected, enabling GPU support"
    elif command_exists rocm-smi && rocm-smi --showdevice 2>/dev/null | grep -q "GPU"; then
        GPU_SUPPORT=true
        log_info "AMD GPU detected, enabling GPU support"
    fi
    
    if [ ! -f "$LLM_MODEL_PATH" ]; then
        log_warning "Default LLM model not found at: $LLM_MODEL_PATH"
        for dir in \
            "$HOME/.cache/huggingface/hub" \
            "$HOME/.cache/models" \
            "$HOME/.local/share/ollama/models" \
            "./models"; do
            if [ -d "$dir" ]; then
                found_model=$(find "$dir" -name "*.gguf" -type f 2>/dev/null | head -n 1)
                if [ -n "$found_model" ]; then
                    LLM_MODEL_PATH="$found_model"
                    log_info "Found alternative LLM model at: $LLM_MODEL_PATH"
                    break
                fi
            fi
        done
    fi
}

check_python_version() {
    if ! command_exists "$PYTHON_CMD"; then
        error_exit "$PYTHON_CMD is required but not found. Please install Python $MIN_PYTHON_VERSION+." "python_version"
    fi

    CURRENT_VERSION=$($PYTHON_CMD -c 'import sys; print(f"{sys.version_info.major}.{sys.version_info.minor}")')
    REQUIRED_MAJOR=$(echo $MIN_PYTHON_VERSION | cut -d. -f1)
    REQUIRED_MINOR=$(echo $MIN_PYTHON_VERSION | cut -d. -f2)
    CURRENT_MAJOR=$(echo $CURRENT_VERSION | cut -d. -f1)
    CURRENT_MINOR=$(echo $CURRENT_VERSION | cut -d. -f2)

    if [ "$CURRENT_MAJOR" -lt "$REQUIRED_MAJOR" ] || { [ "$CURRENT_MAJOR" -eq "$REQUIRED_MAJOR" ] && [ "$CURRENT_MINOR" -lt "$REQUIRED_MINOR" ]; }; then
        error_exit "Python version $MIN_PYTHON_VERSION or higher required, but found $CURRENT_VERSION." "python_version"
    fi
    log_info "Python version $CURRENT_VERSION found."
}

is_port_free() {
    PORT_NUM=$1
    if command_exists ss; then
        ss -tuln | grep -q ":$PORT_NUM " && return 1
    elif command_exists netstat; then
        netstat -tuln | grep -q ":$PORT_NUM " && return 1
    elif command_exists lsof; then
        lsof -i ":$PORT_NUM" &> /dev/null && return 1
    else
        log_warning "Cannot check port $PORT_NUM - 'ss', 'netstat', or 'lsof' not found. Assuming free."
        return 0
    fi
    return 0
}

is_process_running_on_port() {
    PORT_NUM=$1
    if command_exists lsof; then
        PROCESS_INFO=$(lsof -i ":$PORT_NUM" -sTCP:LISTEN -n -P 2>/dev/null)
        if [ -n "$PROCESS_INFO" ]; then
            PROCESS_NAME=$(echo "$PROCESS_INFO" | tail -n 1 | awk '{print $1}')
            PROCESS_PID=$(echo "$PROCESS_INFO" | tail -n 1 | awk '{print $2}')
            echo "$PROCESS_NAME (PID $PROCESS_PID)"
            return 0
        fi
    fi
    return 1
}

find_free_port() {
    START_PORT=$1
    DESCRIPTION=$2
    PORT=$START_PORT
    while ! is_port_free "$PORT"; do
        PROCESS_INFO=$(is_process_running_on_port "$PORT")
        if [ $? -eq 0 ]; then
            log_warning "Port $PORT ($DESCRIPTION) is in use by $PROCESS_INFO."
            if [ "$FORCE_MODE" = true ]; then
                log_info "Force mode enabled. Attempting to kill process $PROCESS_INFO..."
                PID=$(echo "$PROCESS_INFO" | grep -o 'PID [0-9]*' | awk '{print $2}')
                kill -9 "$PID" 2>/dev/null && log_info "Killed process on port $PORT" || log_warning "Failed to kill process. Finding next port..."
            fi
        fi
        if ! is_port_free "$PORT"; then
            PORT=$((PORT + 1))
            if [ "$PORT" -gt 65535 ]; then
                error_exit "No free ports available up to 65535 for $DESCRIPTION." "port_conflict"
            fi
        fi
    done
    if [ "$PORT" != "$START_PORT" ]; then
        log_info "Resolved conflict: Using free port $PORT instead of $START_PORT for $DESCRIPTION."
    else
        log_info "Port $PORT ($DESCRIPTION) is free."
    fi
    echo "$PORT"
}

create_backup() {
    local file="$1"
    if [ -f "$file" ]; then
        local backup_path="$BACKUP_DIR/$(basename "$file").bak-$SETUP_ID"
        mkdir -p "$BACKUP_DIR"
        cp "$file" "$backup_path"
        log_debug "Created backup of $file at $backup_path"
    fi
}

update_config_port() {
    CONFIG_FILE="$1"
    SECTION="$2"
    KEY="$3"
    NEW_PORT="$4"
    
    create_backup "$CONFIG_FILE"
    
    if command_exists jq; then
        local tmp_file=$(mktemp)
        jq ".$SECTION.$KEY = $NEW_PORT" "$CONFIG_FILE" > "$tmp_file"
        if [ $? -eq 0 ] && [ -s "$tmp_file" ]; then
            mv "$tmp_file" "$CONFIG_FILE"
            log_info "Updated $CONFIG_FILE: Set $SECTION.$KEY to $NEW_PORT (using jq)"
        else
            rm -f "$tmp_file"
            $PYTHON_CMD -c "import json; f = open('$CONFIG_FILE', 'r'); data = json.load(f); f.close(); data['$SECTION']['$KEY'] = $NEW_PORT; f = open('$CONFIG_FILE', 'w'); json.dump(data, f, indent=2); f.write('\n'); f.close()" || error_exit "Failed to update $CONFIG_FILE with $SECTION.$KEY=$NEW_PORT" "config_update"
            log_info "Updated $CONFIG_FILE: Set $SECTION.$KEY to $NEW_PORT (using Python)"
        fi
    else
        $PYTHON_CMD -c "import json; f = open('$CONFIG_FILE', 'r'); data = json.load(f); f.close(); data['$SECTION']['$KEY'] = $NEW_PORT; f = open('$CONFIG_FILE', 'w'); json.dump(data, f, indent=2); f.write('\n'); f.close()" || error_exit "Failed to update $CONFIG_FILE with $SECTION.$KEY=$NEW_PORT" "config_update"
        log_info "Updated $CONFIG_FILE: Set $SECTION.$KEY to $NEW_PORT (using Python)"
    fi
}

check_system_dependencies() {
    log_info "Checking system dependencies..."
    
    MISSING_ESSENTIALS=()
    if ! command_exists gcc && ! command_exists clang; then
        MISSING_ESSENTIALS+=("C compiler (gcc or clang)")
    fi
    if ! command_exists make; then
        MISSING_ESSENTIALS+=("make")
    fi
    if ! command_exists cmake; then
        MISSING_ESSENTIALS+=("cmake")
    fi
    if ! command_exists git; then
        MISSING_ESSENTIALS+=("git")
    fi
    
    if [ ${#MISSING_ESSENTIALS[@]} -gt 0 ]; then
        log_error "Missing essential build dependencies:"
        for dep in "${MISSING_ESSENTIALS[@]}"; do
            log_error "- $dep"
        done
        if [ "$FORCE_MODE" = false ]; then
            error_exit "Please install missing dependencies."
        else
            log_warning "Continuing despite missing dependencies (--force mode)."
        fi
    else
        log_info "All essential build dependencies are installed."
    fi
}

check_python_packages() {
    log_info "Checking for essential Python packages..."
    
    ESSENTIAL_PKGS=("pip" "setuptools" "wheel" "venv")
    MISSING_PKGS=()
    
    for pkg in "${ESSENTIAL_PKGS[@]}"; do
        if ! $PYTHON_CMD -c "import $pkg" 2>/dev/null; then
            MISSING_PKGS+=("$pkg")
        fi
    done
    
    if [ ${#MISSING_PKGS[@]} -gt 0 ]; then
        log_warning "Missing essential Python packages: ${MISSING_PKGS[*]}"
        if [ "$FORCE_MODE" = true ]; then
            log_info "Attempting to install missing Python packages..."
            if [[ " ${MISSING_PKGS[*]} " =~ " pip " ]]; then
                $PYTHON_CMD -m ensurepip --upgrade || error_exit "Failed to bootstrap pip." "pip"
            fi
            for pkg in "${MISSING_PKGS[@]}"; do
                if [ "$pkg" != "pip" ]; then
                    $PYTHON_CMD -m pip install --user "$pkg" || log_warning "Failed to install $pkg"
                fi
            done
        else
            error_exit "Please install missing Python packages: ${MISSING_PKGS[*]}" "pip"
        fi
    else
        log_info "All essential Python packages are installed."
    fi
}

create_startup_script() {
    local startup_script="$PROJECT_FULL_PATH/start_kaleidoscope.sh"
    
    cat > "$startup_script" << EOF
#!/bin/bash
# Kaleidoscope AI Startup Script
# Generated on $(date)

PROJECT_DIR="$PROJECT_FULL_PATH"
VENV_DIR="$VENV_DIR"
CONFIG_FILE="$CONFIG_FILE"
LLM_PROVIDER="$LLM_PROVIDER"
LLM_MODEL_PATH="$LLM_MODEL_PATH"
LLM_PORT="$LLM_PORT"
WEB_PORT="$WEB_PORT"
API_PORT="$API_PORT"
VIS_PORT="$VIS_PORT"
QUANTUM_PORT="$QUANTUM_PORT"
LLAMA_SERVER_CMD="$LLAMA_SERVER_CMD"

SESSION_NAME="kaleidoscope_\$(date +%s)"

start_services() {
    cd "\$PROJECT_DIR" || { echo "Error: Cannot change to project directory"; exit 1; }
    source "\$VENV_DIR/bin/activate" || { echo "Error: Cannot activate virtual environment"; exit 1; }
    
    if command -v tmux &> /dev/null; then
        echo "Starting services in new tmux session..."
        tmux new-session -d -s "\$SESSION_NAME" -n "kaleidoscope"
        
        if [ "\$LLM_PROVIDER" == "llamacpp_api" ]; then
            tmux split-window -h -t "\$SESSION_NAME" "\$LLAMA_SERVER_CMD -m '\$LLM_MODEL_PATH' -c 2048 --host 127.0.0.1 --port \$LLM_PORT; read -p 'Press Enter to close'"
            sleep 2
        fi
        
        tmux split-window -v -t "\$SESSION_NAME" "cd '\$PROJECT_DIR' && source '\$VENV_DIR/bin/activate' && python quantum/quantum_server.py --port \$QUANTUM_PORT; read -p 'Press Enter to close'"
        sleep 2
        
        tmux select-pane -t "\$SESSION_NAME:0.0"
        tmux send-keys -t "\$SESSION_NAME:0.0" "cd '\$PROJECT_DIR' && source '\$VENV_DIR/bin/activate' && python main.py --port \$WEB_PORT --api-port \$API_PORT --vis-port \$VIS_PORT" C-m
        
        tmux attach-session -t "\$SESSION_NAME"
    else
        echo "tmux not available. Starting services manually..."
        if [ "\$LLM_PROVIDER" == "llamacpp_api" ]; then
            echo "Start Llama.cpp server in a new terminal:"
            echo "\$LLAMA_SERVER_CMD -m '\$LLM_MODEL_PATH' -c 2048 --host 127.0.0.1 --port \$LLM_PORT"
        fi
        echo "Start Quantum server in a new terminal:"
        echo "cd '\$PROJECT_DIR' && source '\$VENV_DIR/bin/activate' && python quantum/quantum_server.py --port \$QUANTUM_PORT"
        python main.py --port "\$WEB_PORT" --api-port "\$API_PORT" --vis-port "\$VIS_PORT"
    fi
}

if [ "\$1" == "--help" ] || [ "\$1" == "-h" ]; then
    echo "Kaleidoscope AI Startup Script"
    echo "Usage: \$0 [OPTIONS]"
    echo "Options:"
    echo "  --help, -h      Show this help message"
    echo "  --restart       Kill and restart all services"
    echo "  --stop          Stop all services"
    exit 0
fi

if [ "\$1" == "--restart" ]; then
    pkill -f "python.*quantum/quantum_server.py" || true
    pkill -f "\$LLAMA_SERVER_CMD" || true
    pkill -f "python.*main.py" || true
    if command -v tmux &> /dev/null && tmux has-session -t "\$SESSION_NAME" 2>/dev/null; then
        tmux kill-session -t "\$SESSION_NAME"
    fi
    sleep 2
    start_services
    exit 0
fi

if [ "\$1" == "--stop" ]; then
    pkill -f "python.*quantum/quantum_server.py" || true
    pkill -f "\$LLAMA_SERVER_CMD" || true
    pkill -f "python.*main.py" || true
    if command -v tmux &> /dev/null && tmux has-session -t "\$SESSION_NAME" 2>/dev/null; then
        tmux kill-session -t "\$SESSION_NAME"
    fi
    echo "All processes stopped."
    exit 0
fi

start_services
EOF

    chmod +x "$startup_script"
    log_info "Created startup script: $startup_script"
}
